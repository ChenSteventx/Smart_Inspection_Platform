2. MySQL数据库配置  高优先级
---------------------------

2.1 数据库表结构
需要在MySQL中创建以下表：

数据库名：school
表1：特种设备安全法规
表2：热搜

SQL创建语句：
`sql
-- 创建数据库
CREATE DATABASE IF NOT EXISTS school CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE school;

-- 创建特种设备安全法规表
CREATE TABLE IF NOT EXISTS 特种设备安全法规 (
    CID varchar(255) DEFAULT NULL,
    Cname varchar(255) DEFAULT NULL,
    Cnumber varchar(255) DEFAULT NULL,
    Cclass varchar(255) DEFAULT NULL,
    Cparty varchar(255) DEFAULT NULL,
    ClawRe varchar(255) DEFAULT NULL,
    CP varchar(255) DEFAULT NULL,
    CP1 varchar(255) DEFAULT NULL,
    CP2 varchar(255) DEFAULT NULL,
    CP3 varchar(255) DEFAULT NULL,
    CP4 varchar(255) DEFAULT NULL,
    Ckey varchar(255) DEFAULT NULL,
    Cmain text,
    Cindex text,
    CB1 varchar(255) DEFAULT NULL,
    CB2 varchar(255) DEFAULT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建热搜表
CREATE TABLE IF NOT EXISTS 热搜 (
    字段 varchar(255) DEFAULT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
`
